### 1.3 条件解析引擎

在`server/webhook/index.ts`中实现条件解析引擎：

```typescript
// 条件类型定义
interface TriggerCondition {
  field: string;
  operator: 'eq' | 'ne' | 'gt' | 'lt' | 'gte' | 'lte' | 'in' | 'between';
  value: any;
  logicOperator?: 'AND' | 'OR';
}

// 条件评估函数
function evaluateCondition(condition: TriggerCondition, context: Record<string, any>): boolean {
  const fieldValue = context[condition.field];
  
  switch (condition.operator) {
    case 'eq': return fieldValue === condition.value;
    case 'ne': return fieldValue !== condition.value;
    case 'gt': return fieldValue > condition.value;
    case 'lt': return fieldValue < condition.value;
    case 'gte': return fieldValue >= condition.value;
    case 'lte': return fieldValue <= condition.value;
    case 'in': return Array.isArray(condition.value) && condition.value.includes(fieldValue);
    case 'between': 
      return Array.isArray(condition.value) && 
             fieldValue >= condition.value[0] && 
             fieldValue <= condition.value[1];
    default: return true;
  }
}

// 评估所有条件
function evaluateAllConditions(conditions: TriggerCondition[], context: Record<string, any>): boolean {
  if (!conditions || conditions.length === 0) return true;
  
  let result = evaluateCondition(conditions[0], context);
  
  for (let i = 1; i < conditions.length; i++) {
    const condition = conditions[i];
    const conditionResult = evaluateCondition(condition, context);
    
    if (condition.logicOperator === 'OR') {
      result = result || conditionResult;
    } else {
      result = result && conditionResult;
    }
  }
  
  return result;
}
```

```typescript
// server/db.ts 中添加
export async function getItemDependencies(configId: number) {
  return await db.select()
    .from(annualPlanningDependencies)
    .where(eq(annualPlanningDependencies.configId, configId));
}

export async function addItemDependency(data: {
  configId: number;
  sourceItemId: number;
  targetItemId: number;
  dependencyType?: string;
  lagDays?: number;
  createdBy?: number;
}) {
  // 检查循环依赖
  const hasCycle = await checkCyclicDependency(data.configId, data.sourceItemId, data.targetItemId);
  if (hasCycle) {
    throw new Error('检测到循环依赖，无法添加');
  }
  
  return await db.insert(annualPlanningDependencies).values(data);
}

export async function removeItemDependency(id: number) {
  return await db.delete(annualPlanningDependencies)
    .where(eq(annualPlanningDependencies.id, id));
}

// 循环依赖检测（DFS算法）
async function checkCyclicDependency(configId: number, sourceId: number, targetId: number): Promise<boolean> {
  const dependencies = await getItemDependencies(configId);
  const graph = new Map<number, number[]>();
  
  // 构建邻接表
  for (const dep of dependencies) {
    if (!graph.has(dep.sourceItemId)) {
      graph.set(dep.sourceItemId, []);
    }
    graph.get(dep.sourceItemId)!.push(dep.targetItemId);
  }
  
  // 添加新边
  if (!graph.has(sourceId)) {
    graph.set(sourceId, []);
  }
  graph.get(sourceId)!.push(targetId);
  
  // DFS检测环
  const visited = new Set<number>();
  const recursionStack = new Set<number>();
  
  function dfs(node: number): boolean {
    visited.add(node);
    recursionStack.add(node);
    
    const neighbors = graph.get(node) || [];
    for (const neighbor of neighbors) {
      if (!visited.has(neighbor)) {
        if (dfs(neighbor)) return true;
      } else if (recursionStack.has(neighbor)) {
        return true;
      }
    }
    
    recursionStack.delete(node);
    return false;
  }
  
  for (const node of graph.keys()) {
    if (!visited.has(node) && dfs(node)) {
      return true;
    }
  }
  
  return false;
}
```

```typescript
// 依赖连线组件
function DependencyLines({ items, dependencies, containerRef }: {
  items: AnnualPlanningItem[];
  dependencies: Dependency[];
  containerRef: React.RefObject<HTMLDivElement>;
}) {
  const [lines, setLines] = useState<LineData[]>([]);
  
  useEffect(() => {
    if (!containerRef.current) return;
    
    const newLines: LineData[] = [];
    
    for (const dep of dependencies) {
      const sourceEl = containerRef.current.querySelector(`[data-item-id="${dep.sourceItemId}"]`);
      const targetEl = containerRef.current.querySelector(`[data-item-id="${dep.targetItemId}"]`);
      
      if (sourceEl && targetEl) {
        const sourceRect = sourceEl.getBoundingClientRect();
        const targetRect = targetEl.getBoundingClientRect();
        const containerRect = containerRef.current.getBoundingClientRect();
        
        newLines.push({
          id: dep.id,
          x1: sourceRect.right - containerRect.left,
          y1: sourceRect.top + sourceRect.height / 2 - containerRect.top,
          x2: targetRect.left - containerRect.left,
          y2: targetRect.top + targetRect.height / 2 - containerRect.top,
          type: dep.dependencyType
        });
      }
    }
    
    setLines(newLines);
  }, [items, dependencies]);
  
  return (
    <svg className="absolute inset-0 pointer-events-none" style={{ overflow: 'visible' }}>
      <defs>
        <marker id="arrowhead" markerWidth="10" markerHeight="7" refX="9" refY="3.5" orient="auto">
          <polygon points="0 0, 10 3.5, 0 7" fill="#f97316" />
        </marker>
      </defs>
      {lines.map(line => (
        <path
          key={line.id}
          d={`M ${line.x1} ${line.y1} C ${line.x1 + 30} ${line.y1}, ${line.x2 - 30} ${line.y2}, ${line.x2} ${line.y2}`}
          stroke="#f97316"
          strokeWidth="2"
          fill="none"
          markerEnd="url(#arrowhead)"
        />
      ))}
    </svg>
  );
}
```

```typescript
// server/db.ts 中添加
export async function saveRuleVersion(ruleId: number, ruleData: any, changedBy?: number, changeSummary?: string) {
  // 获取当前最大版本号
  const [maxVersion] = await db.select({ max: sql`MAX(version_number)` })
    .from(costAlertRuleVersions)
    .where(eq(costAlertRuleVersions.ruleId, ruleId));
  
  const newVersionNumber = (maxVersion?.max || 0) + 1;
  
  return await db.insert(costAlertRuleVersions).values({
    ruleId,
    versionNumber: newVersionNumber,
    ruleData: JSON.stringify(ruleData),
    changeSummary,
    changedBy
  });
}

export async function getRuleVersions(ruleId: number) {
  return await db.select()
    .from(costAlertRuleVersions)
    .where(eq(costAlertRuleVersions.ruleId, ruleId))
    .orderBy(desc(costAlertRuleVersions.versionNumber));
}

export async function getRuleVersion(ruleId: number, versionNumber: number) {
  const [version] = await db.select()
    .from(costAlertRuleVersions)
    .where(and(
      eq(costAlertRuleVersions.ruleId, ruleId),
      eq(costAlertRuleVersions.versionNumber, versionNumber)
    ));
  return version;
}

export async function rollbackRuleToVersion(ruleId: number, versionNumber: number, userId?: number) {
  const version = await getRuleVersion(ruleId, versionNumber);
  if (!version) throw new Error('版本不存在');
  
  const ruleData = JSON.parse(version.ruleData);
  
  // 更新规则
  await db.update(costAlertRules)
    .set(ruleData)
    .where(eq(costAlertRules.id, ruleId));
  
  // 保存新版本（回滚操作也记录为新版本）
  await saveRuleVersion(ruleId, ruleData, userId, `回滚到版本 ${versionNumber}`);
  
  return { success: true };
}
```

```typescript
// 版本对比工具函数
function compareRuleVersions(oldVersion: RuleVersionData, newVersion: RuleVersionData): VersionDiff[] {
  const diffs: VersionDiff[] = [];
  const allKeys = new Set([...Object.keys(oldVersion), ...Object.keys(newVersion)]);
  
  for (const key of allKeys) {
    const oldValue = oldVersion[key];
    const newValue = newVersion[key];
    
    if (JSON.stringify(oldValue) !== JSON.stringify(newValue)) {
      diffs.push({
        field: key,
        fieldLabel: getFieldLabel(key),
        oldValue: formatValue(oldValue),
        newValue: formatValue(newValue),
        changeType: oldValue === undefined ? 'added' : 
                    newValue === undefined ? 'removed' : 'modified'
      });
    }
  }
  
  return diffs;
}

interface VersionDiff {
  field: string;
  fieldLabel: string;
  oldValue: string;
  newValue: string;
  changeType: 'added' | 'removed' | 'modified';
}
```
